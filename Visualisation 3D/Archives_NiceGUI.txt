"""

from nicegui import ui

# État global du robot
robot_state = {
    "x": 0.0,
    "y": 0.0,
    "z": 1.0,  # Position initiale en Z (au-dessus du sol)
    "rz": 0.0,  # Rotation du 4ème axe (en radians)
}

# Limites de déplacement sécurisées
LIMITS = {
    "x": (-0.8, 0.8),
    "y": (-0.8, 0.8),
    "z": (0.3, 1.8),
}

# Hauteur fixe de la base supérieure du robot
BASE_Z = 2.0

# Éléments graphiques 3D globaux
end_effector = None
link_lines = []
scene = None

# Variable pour suivre l'axe et le sens du mouvement actif (maintenu)
active_jog = {"axis": None, "delta": 0.0}


def update_robot_graphics():
    """Met à jour la position de l'effecteur et redessine les bras de liaison"""
    global end_effector, link_lines, scene

    if not end_effector or not scene:
        return

    x = robot_state["x"]
    y = robot_state["y"]
    z = robot_state["z"]
    rz = robot_state["rz"]

    # 1. Déplacer et orienter correctement l'effecteur mobile
    for child in list(end_effector.children):
        child.delete()
    with end_effector:
        # Plateau horizontal de la plateforme mobile
        ui.scene.box(0.4, 0.1, 0.4).material("#2c3e50").rotate(1.5708, 0, 0)
        # 4ème axe : cylindre vertical orienté selon Z
        ui.scene.cylinder(0.05, 0.05, 0.5).material("#e74c3c").rotate(
            1.5708, 0, rz
        ).move(0, 0, 0)

    end_effector.move(x, y, z)

    # 2. Redessiner les 3 bras cinématiques de liaison
    base_joints = [
        (0.6, 0.0, BASE_Z),
        (-0.3, 0.52, BASE_Z),
        (-0.3, -0.52, BASE_Z),
    ]

    for line in link_lines:
        line.delete()
    link_lines.clear()

    for bx, by, bz in base_joints:
        control1 = [
            bx + (x - bx) / 3,
            by + (y - by) / 3,
            bz + (z - bz) / 3,
        ]
        control2 = [
            bx + 2 * (x - bx) / 3,
            by + 2 * (y - by) / 3,
            bz + 2 * (z - bz) / 3,
        ]
        line = scene.curve([bx, by, bz], control1, control2, [x, y, z]).material(
            "#f1c40f"
        )
        link_lines.append(line)

    # Mettre à jour l'affichage des coordonnées
    coord_label.text = f"X: {x:.2f} m | Y: {y:.2f} m | Z: {z:.2f} m | Rz: {rz:.2f} rad"


def apply_jog():
    """Fonction appelée à intervalle régulier (timer) si un bouton est maintenu enfoncé"""
    axis = active_jog["axis"]
    delta = active_jog["delta"]

    if axis is None or delta == 0.0:
        return

    if axis in ["x", "y", "z"]:
        new_val = robot_state[axis] + delta
        if LIMITS[axis][0] <= new_val <= LIMITS[axis][1]:
            robot_state[axis] = new_val
    elif axis == "rz":
        robot_state["rz"] += delta

    update_robot_graphics()


def start_jog(axis: str, delta: float):
    """Déclenche le mouvement continu"""
    active_jog["axis"] = axis
    active_jog["delta"] = delta


def stop_jog():
    """Arrête le mouvement continu"""
    active_jog["axis"] = None
    active_jog["delta"] = 0.0


def reset_robot():
    """Réinitialise le robot à sa position HOME"""
    stop_jog()
    robot_state["x"] = 0.0
    robot_state["y"] = 0.0
    robot_state["z"] = 1.0
    robot_state["rz"] = 0.0
    ui.notify("Robot réinitialisé en position HOME", type="info")
    update_robot_graphics()


# --- INTERFACE UTILISATEUR NICEGUI ---

ui.markdown("# 🤖 Simulation Live - Robot Delta 4 Axes")

with ui.row().classes("w-full items-start gap-4"):

    # --- COLONNE DE GAUCHE : SCÈNE 3D ---
    with ui.card().classes("flex-grow bg-grey-2 p-0"):
        with ui.scene(width=800, height=600) as sc:
            scene = sc
            # Caméra positionnée pour voir le sol et le robot en hauteur
            sc.move_camera(
                x=3.5,
                y=-3.5,
                z=2.0,
                look_at_x=0,
                look_at_y=0,
                look_at_z=1.0,
            )

            # Sol de référence (Grille à Z = 0)
            # Structure fixe du Robot Delta (en haut à BASE_Z)
            with ui.scene.group().move(0, 0, BASE_Z):
                ui.scene.cylinder(0.9, 0.9, 0.1).material("#7f8c8d").rotate(
                    1.5708, 0, 0
                )
                for bx, by in [(0.6, 0.0), (-0.3, 0.52), (-0.3, -0.52)]:
                    ui.scene.cylinder(0.04, 0.04, BASE_Z).material(
                        "#bdc3c7"
                    ).rotate(1.5708, 0, 0).move(bx, by, -BASE_Z / 2)

            # Groupe de l'effecteur mobile
            end_effector = ui.scene.group()

    # --- COLONNE DE DROITE : TEACH PENDANT (CONTRÔLE) ---
    with ui.card().classes("w-80 p-4 flex flex-col gap-4"):
        ui.label("Teach Pendant (Jogging Continu)").classes(
            "text-lg font-bold text-primary"
        )

        coord_label = ui.label("").classes(
            "text-sm font-mono bg-dark text-white p-2 rounded"
        )

        step = 0.025  # Pas plus fin pour une transition fluide en continu


        # Fonction utilitaire pour créer les boutons avec écouteurs de clic maintenu
        def create_jog_button(label, axis, delta, color_class):
            btn = ui.button(label).classes(f"{color_class} text-white flex-1")
            # Événements JS natifs pour détecter le maintien du clic
            btn.on("mousedown", lambda: start_jog(axis, delta))
            btn.on("mouseup", lambda: stop_jog())
            btn.on("mouseleave", lambda: stop_jog())
            return btn


        # Contrôles Axe X
        ui.label("Axe X").classes("font-semibold text-gray-700 mb-[-10px]")
        with ui.row().classes("w-full justify-between"):
            create_jog_button("X -", "x", -step, "bg-red-500")
            create_jog_button("X +", "x", step, "bg-green-500")

        # Contrôles Axe Y
        ui.label("Axe Y").classes("font-semibold text-gray-700 mb-[-10px]")
        with ui.row().classes("w-full justify-between"):
            create_jog_button("Y -", "y", -step, "bg-red-500")
            create_jog_button("Y +", "y", step, "bg-green-500")

        # Contrôles Axe Z
        ui.label("Axe Z").classes("font-semibold text-gray-700 mb-[-10px]")
        with ui.row().classes("w-full justify-between"):
            create_jog_button("Z -", "z", -step, "bg-red-500")
            create_jog_button("Z +", "z", step, "bg-green-500")

        # Contrôles Axe Rz (Rotation)
        ui.label("Rotation 4e Axe (Rz)").classes(
            "font-semibold text-gray-700 mb-[-10px]"
        )
        with ui.row().classes("w-full justify-between"):
            create_jog_button("Rz -", "rz", -0.1, "bg-orange-500")
            create_jog_button("Rz +", "rz", 0.1, "bg-orange-500")

        ui.separator()

        ui.button("🏠 RESET HOME", on_click=reset_robot).classes(
            "w-full bg-blue-600 text-white font-bold"
        )

# Timer global de rafraîchissement (exécuté toutes les 30 millisecondes)
ui.timer(0.03, apply_jog)

# Lancer la mise à jour initiale de la position
update_robot_graphics()

from threading import Lock


# ============================================================
# ÉTAT GLOBAL ET PARTAGÉ DU ROBOT

}

BASE_Z = 2.0


# ============================================================
# SYNCHRONISATION DU ROBOT
# ============================================================

robot_lock = Lock()

# Augmente à chaque modification de la position globale
robot_version = 0

# PAGE ACCUEIL

@ui.page("/")
def home_page():

                "hover:bg-blue-700 no-underline"

# ============================================================
# PAGE ROBOT
# ============================================================

@ui.page("/robot")
def robot_page():

    global robot_version

    # ========================================================
    # VARIABLES LOCALES À CE NAVIGATEUR
    # ========================================================

    # Chaque utilisateur possède sa propre scène
    scene = None
    end_effector = None
    coord_label = None

    # Les trois courbes jaunes de CETTE scène
    link_lines = []

    # Jogging propre à cet utilisateur
    active_jog = {
        "axis": None,
        "delta": 0.0,
    }

    # Version du robot actuellement affichée
    last_version = -1


    # ========================================================
    # MISE À JOUR DE LA SCÈNE 3D
    # ========================================================

    def update_robot_graphics():

        nonlocal last_version

        if (
            scene is None
            or end_effector is None
            or coord_label is None
        ):
            return

        # ----------------------------------------------------
        # Lire l'état global du robot
        # ----------------------------------------------------

        with robot_lock:

            x = robot_state["x"]
            y = robot_state["y"]
            z = robot_state["z"]
            rz = robot_state["rz"]

            current_version = robot_version


        # ====================================================
        # EFFECTEUR MOBILE
        # ====================================================

        # Supprimer l'ancien contenu
        for child in list(end_effector.children):
            child.delete()

        # Recréer l'effecteur exactement comme dans TEST2
        with end_effector:

            # Plateau horizontal
            ui.scene.box(
                0.4,
                0.1,
                0.4
            ).material(
                "#2c3e50"
            ).rotate(
                1.5708,
                0,
                0
            )

            # =================================================
            # 4ème AXE
            # =================================================

            ui.scene.cylinder(
                0.05,
                0.05,
                0.5
            ).material(
                "#e74c3c"
            ).rotate(
                1.5708,
                0,
                rz
            ).move(
                0,
                0,
                0
            )

        # Positionner l'effecteur
        end_effector.move(x, y, z)
        # ====================================================
        # POINT CENTRAL DE L'EFFECTEUR
        # ====================================================

        
        scene.sphere(
            0.06,
            12,
            8
        ).material("#00ff00").move(x, y, z)


        # ====================================================
        # TROIS LIAISONS JAUNES
        # ====================================================

        base_joints = [
            (0.6, 0.0, BASE_Z),
            (-0.3, 0.52, BASE_Z),
            (-0.3, -0.52, BASE_Z),
        ]

        # Supprimer les anciennes courbes
        for line in link_lines:
            line.delete()

        link_lines.clear()


        # Recréer les trois courbes
        for bx, by, bz in base_joints:

            control1 = [
                bx + (x - bx) / 3,
                by + (y - by) / 3,
                bz + (z - bz) / 3,
            ]

            control2 = [
                bx + 2 * (x - bx) / 3,
                by + 2 * (y - by) / 3,
                bz + 2 * (z - bz) / 3,
            ]

            line = scene.curve(
                [bx, by, bz],
                control1,
                control2,
                [x, y, z],
            ).material(
                "#f1c40f"
            )

            link_lines.append(line)


        # ====================================================
        # COORDONNÉES
        # ====================================================

        coord_label.text = (
            f"X: {x:.2f} m | "
            f"Y: {y:.2f} m | "
            f"Z: {z:.2f} m | "
            f"Rz: {rz:.2f} rad"
        )

        # Mémoriser la version affichée
        last_version = current_version


    # ========================================================
    # DÉPLACER LE ROBOT GLOBAL
    # ========================================================

    def move_robot(axis, delta):

        global robot_version

        with robot_lock:

            # ------------------------------------------------
            # X / Y / Z
            # ------------------------------------------------

            if axis in ("x", "y", "z"):

                new_value = (
                    robot_state[axis]
                    + delta
                )

                minimum, maximum = LIMITS[axis]

                if not (
                    minimum
                    <= new_value
                    <= maximum
                ):
                    return

                robot_state[axis] = new_value

            # ------------------------------------------------
            # RZ
            # ------------------------------------------------

            elif axis == "rz":

                robot_state["rz"] += delta

            else:
                return

            # Signaler que le robot a changé
            robot_version += 1


    # ========================================================
    # JOG CONTINU
    # ========================================================

    def apply_jog():

        axis = active_jog["axis"]
        delta = active_jog["delta"]

        if axis is None:
            return

        if delta == 0.0:
            return

        # Modifier le robot partagé
        move_robot(
            axis,
            delta
        )

        # Mise à jour immédiate de CET utilisateur
        update_robot_graphics()


    # ========================================================
    # DÉMARRER JOG
    # ========================================================

    def start_jog(axis, delta):

        active_jog["axis"] = axis
        active_jog["delta"] = delta


    # ========================================================
    # ARRÊTER JOG
    # ========================================================

    def stop_jog():

        active_jog["axis"] = None
        active_jog["delta"] = 0.0


    # ========================================================
    # RESET HOME
    # ========================================================

    def reset_robot():

        global robot_version

        stop_jog()

        with robot_lock:

            robot_state["x"] = 0.0
            robot_state["y"] = 0.0
            robot_state["z"] = 1.0
            robot_state["rz"] = 0.0

            robot_version += 1

        ui.notify(
            "Robot réinitialisé en position HOME",
            type="info"

def clamp(value, minimum, maximum):
    """Limite une valeur entre minimum et maximum."""
    return max(minimum, min(maximum, value))

    """

def move_robot(axis, delta):
    """
    Modifie l'état global partagé du robot.

    Si l'utilisateur A déplace le robot,
    l'utilisateur B verra également le déplacement.
    """

    global robot_version

    with robot_lock:

        # -------------------------------
        # AXE X
        # -------------------------------

        if axis == "x":

            robot_state["x"] = clamp(
                robot_state["x"] + delta,
                LIMITS["x"][0],
                LIMITS["x"][1],
            )

        # -------------------------------
        # AXE Y
        # -------------------------------

        elif axis == "y":

            robot_state["y"] = clamp(
                robot_state["y"] + delta,
                LIMITS["y"][0],
                LIMITS["y"][1],
            )

        # -------------------------------
        # AXE Z
        # -------------------------------

        elif axis == "z":

            robot_state["z"] = clamp(
                robot_state["z"] + delta,
                LIMITS["z"][0],
                LIMITS["z"][1],
            )

        # -------------------------------
        # ROTATION RZ
        # -------------------------------

        elif axis == "rz":

            robot_state["rz"] += delta

            if robot_state["rz"] > math.pi:
                robot_state["rz"] -= 2 * math.pi

            if robot_state["rz"] < -math.pi:
                robot_state["rz"] += 2 * math.pi

        # Nouvelle version de l'état partagé
        robot_version += 1
