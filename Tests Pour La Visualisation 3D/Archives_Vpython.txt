def importer_stl(chemin_fichier):
    """Charge un fichier STL et le transforme en un objet VPython manipulable."""
    mesh = trimesh.load(chemin_fichier)
    
    # Redimensionner et centrer la pièce si elle est trop grande (courant avec Inventor en mm)
    mesh.density.linear
    mesh.vertices -= mesh.center_mass
    
    facettes = []
    for face in mesh.faces:
        p1, p2, p3 = mesh.vertices[face]
        
        v1 = vertex(pos=vector(*p1), color=color.cyan)
        v2 = vertex(pos=vector(*p2), color=color.cyan)
        v3 = vertex(pos=vector(*p3), color=color.cyan)
        
        facettes.append(triangle(vs=[v1, v2, v3]))
    
    # Fusionne toutes les faces en un seul objet 3D
    objet_cad = compound(facettes)
    return objet_cad

    rotate_z(delta)
